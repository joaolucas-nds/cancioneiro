/* ---- Importar varios .json (0.22.0, i-057, DEC-040 — spec meta/specs/261116-varios-json.md) ----
   Um arquivo segue o caminho de sempre (importFromFile, readTracksFile). Varios abrem a
   janela «Importar N arquivos»; soltos na lateral, fora de um album, viram acervos
   guardados direto (tambem um so). Acervo novo e guardado SEM trocar o aberto; sem
   nenhum aberto, abre o primeiro. */

/** Le um arquivo como texto (Promise). */
function readFileText(file) {
  return new Promise((resolve, reject) => {
    const r = new FileReader();
    r.onload = () => resolve(r.result);
    r.onerror = () => reject(r.error || new Error("leitura"));
    r.readAsText(file);
  });
}

/** Le e classifica cada arquivo: {name, file, handle, json, extracted, removed, why}.
    `why` diz por que o arquivo nao serve (e a linha vem desligada); null = serve. */
async function readEntries(entries) {
  const out = [];
  for (const { file, handle } of entries) {
    const e = { name: file.name, file, handle: handle || null, json: null, extracted: false, removed: 0, why: null };
    let text = null;
    try { text = await readFileText(file); } catch (err) { e.why = "não consegui ler o arquivo"; }
    if (text !== null) {
      try {
        ({ json: e.json, extracted: e.extracted } = parseCollectionText(text));
        e.removed = stripInternalKeys(e.json);
        if (!e.json || !Array.isArray(e.json.albums)) { e.json = null; e.why = "não é um acervo do Cancioneiro"; }
      } catch (err) { e.json = null; e.why = "não é um JSON válido"; }
    }
    out.push(e);
  }
  return out;
}

/** O que o arquivo tem, para a linha da janela: «2 álbuns · 14 faixas · 3 fórmulas». */
function entrySummary(json) {
  const albums = json.albums.length;
  const songs = json.albums.reduce((k, a) => k + (a.songs || []).length, 0);
  const refs = importableRefs(json).list.length;

    + (refs ? " · " + nPlural(refs, "fórmula", "fórmulas") : "");
}

DROP_JS = """([sel, files]) => {
  const dt = new DataTransfer();
  files.forEach(([n, t]) => dt.items.add(new File([t], n, { type: 'application/json' })));
  const el = document.querySelector(sel);
  for (const type of ['dragenter', 'dragover', 'drop'])
    el.dispatchEvent(new DragEvent(type, { bubbles: true, cancelable: true, dataTransfer: dt }));
}"""
HOVER_JS = """([sel, n]) => {
  const dt = new DataTransfer();
  for (let i = 0; i < n; i++) dt.items.add(new File(['{}'], 'x' + i + '.json', { type: 'application/json' }));
  const el = document.querySelector(sel);
  for (const type of ['dragenter', 'dragover'])
    el.dispatchEvent(new DragEvent(type, { bubbles: true, cancelable: true, dataTransfer: dt }));
  return document.querySelector('.drop-hint').textContent;
}"""


def coll(n_tracks, n_albums, title, prefix):
    """Acervo sintetico com ids proprios (nao colidem com o demo)."""
    c = make_collection(n_tracks, n_albums)

    for a in c["albums"]:
        a["id"] = prefix + "-" + a["id"]
        for s in a["songs"]:
            s["id"] = prefix + "-" + s["id"]
    return c
